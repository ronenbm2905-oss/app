```js
// Driver contact for a game, quiet by default.
//
// The phone belongs to someone who never gave it to us — the bus company did — so a call
// site that has not thought about who is reading gets the name and nothing else. The three
// places allowed to show the number ask for it explicitly.
export function driverLine(game, withPhone = false) {
  if (!game || game.isHome) return "";
  const name = (game.driverName || "").trim();
  const phone = (game.driverPhone || "").trim();
  if (!withPhone) return name;
  return [name, phone].filter(Boolean).join(" · ");
}
```

במפורש: **החזר את הפיצ'ר. אל תשמור אותו כלל.** מאגר על אדם שלא מסר לנו דבר, שקיים כדי
לשלוח מידע לגורם שכבר מחזיק בו, הוא סיכון ללא תמורה.
